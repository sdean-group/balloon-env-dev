  \left[\frac{\sum_{i\in\kappa(R)}U_i\!\left(W_i\odot

\node[anchor=north,text width=8.40in,align=center,text=muted,font=\capfont]
  at (14.975,28.86) {$\kappa(R)$ selects windows; $U_i$ embeds local predictions;
  $W_i$ downweights window edges.};

\node[anchor=north west,text width=8.55in,align=left,text=navy,
