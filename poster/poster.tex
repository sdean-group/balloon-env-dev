  at (14.975,33.56) {\textbf{Unbounded coordinates}\quad\textcolor{teal}{|}\quad
  \textbf{random access}\quad\textcolor{teal}{|}\quad
  \textbf{same seed + same coordinate = same wind}};

\node[anchor=north west,text width=8.55in,align=left,fill=paleorange,
  rounded corners=.05in,inner sep=.16in,text=ink,font=\smallfont]
  at (10.70,32.20) {\textcolor{navy}{\textbf{One query, without the dense equation}}\\[1mm]
  \textbf{Given:} requested region $R$, diffusion levels $k=K,\ldots,0$, bounded
  denoiser $D_k$, deterministic seed $s$, and conditioning $c$ (coordinates + time).\\[1mm]
  \textbf{1. Cover} $R$ with overlapping $64\!\times\!64$ windows $Q_i$ and include the
  larger parent windows needed at level $k+1$.\\
  \textbf{2. Reuse} every cached parent state $x_{k+1}[Q_i]$; create only missing
  coordinate-indexed noise from $s$.\\
  \textbf{3. Denoise} each window: $y_i=D_k(x_{k+1}[Q_i],c_i)$.\\
  \textbf{4. Blend} overlap with edge taper $w_i$, normalize by total weight, and cache
  the result $x_k[R]$.\\[1mm]
  \textcolor{muted}{$R$: query; $Q_i$: local window; $k$: noise level; $c_i$: local
  condition; $w_i$: edge taper; cache: previously computed state at fixed coordinates.}};
